--Note: You can run queries below on the Prod Server to get the information on what folders need to be created on the newly built DR SQL Server
----------------------------------------------

--View all the folder structures used by a SQL Server
SELECT distinct LEFT(physical_name, LEN(physical_name) - CHARINDEX('\', REVERSE(physical_name)) + 1) as FolderName
from sys.master_files	
order by FolderName

----------------------------------------------
--View all Data files on a SQL Server (for any SQL Server)
select 	
	database_id,
	db_name(database_id) as DBName,
	name as LogicalFileName, 
	physical_name as FileName,
	(size*8)/1024 as SizeMB, 
	max_size as MaxSize,
	growth as FileGrowth,
	is_percent_growth as FileGrowthInPercent,
	type_desc as FileType, 
	state_desc as FileState
from sys.master_files	
order by DBName, Filename
