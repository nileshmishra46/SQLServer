(tr.parent_class = 0)

		INNER JOIN CV3ClientDocHistory AS Hist

		INNER JOIN CV3PatientCareDocument AS pcd

CREATE PROCEDURE SXAHMHealthEventStatusSelPr
(
   @ClientGUID HVCIDdt
)

AS
SET NOCOUNT ON

/*
Confidential and proprietary information of Allscripts Healthcare Solutions, Inc. and/or its affiliates. Authorized
users only. U.S. Government Users: ?Commercial Computer Software.? Subject to full notice set forth herein.
*/

DECLARE @HealthEventStatusDue bit
DECLARE @HealthEventStatusNearDue bit
DECLARE @HealthEventStatusHasSchedule bit

SET @HealthEventStatusDue = 0           -- Nothing due now
SET @HealthEventStatusNearDue = 0       -- Nothing near due
SET @HealthEventStatusHasSchedule = 0   -- Assume no assigned schedule

    -- Used to lookup the visibility of occurrence, based on parent/child combo rules
    DECLARE @schedOccs TABLE ( ScheduledEventOccurrenceID bigint) 
    
    INSERT INTO @schedOccs ( ScheduledEventOccurrenceID )
	SELECT 
	   seo.ScheduledEventOccurrenceID
	FROM 
	   dbo.SXAHMClientEventDisplayFlagTblFn (@ClientGUID) df
	   INNER JOIN SXAHMClientEvent ce ON (ce.ClientEventID = df.ClientEventID)
	   INNER JOIN SXAHMScheduledEventOccurrence seo ON (seo.EventID = ce.EventID and ce.ClientGUID = seo.ClientGUID)
   WHERE
      df.IsDisplayed = 1 and seo.OccurrenceStatusType = 0


    -- Need today's date without the time information
	DECLARE @today DATE 
	SET @today = dbo.SXADBGetEnterpriseNowFn(); 
	
	-- Determine if there is at least one event that is due
	SELECT @HealthEventStatusDue = 
	   ISNULL((SELECT TOP 1 1 
	          FROM SXAHMScheduledEventOccurrence seo
	          INNER JOIN @schedOccs occs ON (seo.ScheduledEventOccurrenceID = occs.scheduledEventOccurrenceID ) 
	          WHERE OccurrenceStatusType = 0 AND ClientGUID = @ClientGUID and seo.ScheduledDateStart <= @today), 0)
	
	-- Determine if there is at least one event that is near due
	SELECT @HealthEventStatusNearDue = 
	   ISNULL((SELECT TOP 1 1
	          FROM SXAHMScheduledEventOccurrence seo
	          INNER JOIN @schedOccs occs ON (seo.ScheduledEventOccurrenceID = occs.scheduledEventOccurrenceID ) 
	          INNER JOIN SXAHMEvent e ON (seo.EventID = e.EventID)
	          WHERE OccurrenceStatusType = 0 AND ClientGUID = @ClientGUID and dbo.SXAHMIsNearDueDateFn(seo.ScheduledDateStart,e.NearDueValue, e.NearDueUOM, @Today)=1), 0)

    -- Determine if there is at least one assigned schedule
	SELECT @HealthEventStatusHasSchedule = 
	   ISNULL( (SELECT TOP 1 1 
                FROM SXAHMClientSchedule s
                WHERE ClientGUID = @ClientGUID AND s.Active = 1), 0)



SELECT @HealthEventStatusDue 'Due', @HealthEventStatusNearDue 'NearDue', @HealthEventStatusHasSchedule 'HasSchedule'
